export const getDocs=async q=>{
  if(store.failList && q[0].name==='infoArticles')throw Error('list failed');
  let docs=Object.keys(store.docs).filter(p=>p.startsWith(q[0].name+'/')).map(path=>snap({path,id:path.split('/')[1]}));
  const condition=q.find(a=>Array.isArray(a)&&a[1]==='==');
  if(condition)docs=docs.filter(d=>d.data()[condition[0]]===condition[2]);
  return {docs};
};
export const runTransaction=async (_db,cb)=>{
  if(store.delay)await new Promise(r=>setTimeout(r,store.delay));
  if(store.failWrite)throw Error('write failed');
  const writes=[];
  const result=await cb({get:async ref=>snap(ref),set:(ref,data)=>writes.push(()=>store.docs[ref.path]=data),delete:ref=>writes.push(()=>delete store.docs[ref.path])});
  writes.forEach(w=>w());store.writes++;return result;
};
